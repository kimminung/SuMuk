// 학 (Crane) — sin⁷ 곡률 날개와 S자 목의 선분 가족
canvas 750 × 1000
ink #141414 width 0.45 alpha 0.65

let wing_span = 0.85 in [0.5, 1.2]
let neck_curl = 1.42 in [1.0, 2.0]

// 날개 펼침 선분 계열 (sin⁷ 날개 포락선)
segments k = 1 ... 2000 {
  from ( -0.6·s(k), -0.1 + 0.55·sin^7(π·s(k)) )
  to   (  0.7·s(k),  0.35·cos(π·s(k)) - 0.2 )
}
where
  s(k) = k/2000

// 목과 머리 선분 계열 (S자 곡선)
segments m = 1 ... 760 {
  from ( 0.05·sin(2π·t(m)), 0.1 + 0.8·t(m) )
  to   ( 0.08·cos(π·t(m)) + 0.02, 0.2 + 0.75·t(m) )
}
where
  t(m) = m/760
